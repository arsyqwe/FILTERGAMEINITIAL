
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.InputSystem;
//using System.Collections.Generic;

[DefaultExecutionOrder(1)]
public class Wall : MonoBehaviour
{
    public float maxX;
    public float minX;
    public float speed;
    public float wallWidth;
    public float wallHeight;
    public float ballRadius;
    public float ballXwidth;
    public float screenRightEdge;
    public float screenLeftEdge;
    Ball[] balls;
  


    void Start()
    {
        balls = FindObjectsOfType<Ball>();
        float halfScreenWidth = Camera.main.aspect * Camera.main.orthographicSize;
        if (balls.Length > 0)
        {
            ballRadius = balls[0].transform.localScale.y / 2f;
            ballXwidth = balls[0].transform.localScale.x / 2f;
        }
        screenRightEdge = halfScreenWidth;
        screenLeftEdge = -halfScreenWidth;
        wallWidth = GetComponent<SpriteRenderer>().bounds.size.x;
        wallHeight = GetComponent<SpriteRenderer>().bounds.size.y;
        Brick brickManager = FindObjectOfType<Brick>();
        if (balls.Length > 0 && brickManager != null)
        {
            brickManager.currentBall = balls[0];
        }
    }

        void Update()
        {

            Debug.Log("Wall");

            maxX = screenRightEdge - (wallWidth / 2f);
            minX = screenLeftEdge + (wallWidth / 2f);

            Vector2 wallPos = transform.position;

            if (Keyboard.current.dKey.isPressed && wallPos.x < maxX)
            {
                wallPos.x += speed * Time.deltaTime;
            }

            if (Keyboard.current.aKey.isPressed && wallPos.x > minX)
            {
                wallPos.x -= speed * Time.deltaTime;
            }

            for(int i = 0; i< balls.Length; i++)
            {
                Ball currentBall = balls[i];
              
            
                float wallFace = wallPos.y + (wallHeight / 2f) + ballRadius;
                float rightPoint = wallPos.x + (wallWidth / 2f) + ballXwidth;
                float leftPoint = wallPos.x - (wallWidth / 2f) - ballXwidth;

                float wallBottom = wallPos.y - (wallHeight / 2f) - ballRadius;
                Vector2 rightBottom = new Vector2(rightPoint, wallBottom);
                Vector2 leftBottom = new Vector2(leftPoint, wallBottom);

                Vector2 rightTopPoint = new Vector2(rightPoint, wallFace);
                Vector2 leftTopPoint = new Vector2(leftPoint, wallFace);


                Vector2 ballCurrentPos = currentBall.transform.position;
                Vector2 ballMovement = currentBall.direction * currentBall.ballSpeed * Time.deltaTime;
                Vector2 ballOldPos = ballCurrentPos - ballMovement;
                Vector2 intersectionPoint;


                if (LineSegmentIntersection(ballOldPos, ballCurrentPos, rightTopPoint, leftTopPoint, out intersectionPoint))
                {
                    if (currentBall.direction.y < 0)
                    {
                        currentBall.direction.y = -currentBall.direction.y;
                        currentBall.transform.position = intersectionPoint;
                    }
                }
                else if (LineSegmentIntersection(ballOldPos, ballCurrentPos, rightTopPoint, rightBottom, out intersectionPoint))
                {
                    if (currentBall.direction.x < 0)
                    {
                        currentBall.direction.x = -currentBall.direction.x;
                        if (currentBall.transform.position.y > wallPos.y)
                        {
                            currentBall.direction.y = -currentBall.direction.y;
                        }

                        
                        currentBall.transform.position = intersectionPoint;
                    }
                }
                else if (LineSegmentIntersection(ballOldPos, ballCurrentPos, leftTopPoint, leftBottom, out intersectionPoint))
                {
                    if (currentBall.direction.x > 0)
                    {
                        currentBall.direction.x = -currentBall.direction.x;
                        if (currentBall.transform.position.y > wallPos.y)
                        {
                            currentBall.direction.y = -currentBall.direction.y;
                        }
                        currentBall.transform.position = intersectionPoint;
                    }
                }   
            }  
            transform.position = wallPos;

            if (Keyboard.current.digit1Key.isPressed)
            {
                Application.targetFrameRate = 3;
            }
            if (Keyboard.current.digit2Key.isPressed)
            {
                Application.targetFrameRate = 15;
            }
            if (Keyboard.current.digit3Key.isPressed)
            {
                Application.targetFrameRate = 30;
            }
            if (Keyboard.current.digit4Key.isPressed)
            {
                Application.targetFrameRate = 60;
            }
        }
        bool LineSegmentIntersection(Vector2 start, Vector2 end, Vector2 start2, Vector2 end2, out Vector2 intersectionPoint)
        {
            Vector2 line1 = end - start;
            Vector2 line2 = end2 - start2;

            float determinant = line1.x * line2.y - line1.y * line2.x;

            if (determinant == 0)
            {
                intersectionPoint = Vector2.zero;
                return false;
            }
            float rat2 = ((start2.x - start.x) * line1.y - (start2.y - start.y) * line1.x) / determinant;
            float rat1 = ((start2.x - start.x) * line2.y - (start2.y - start.y) * line2.x) / determinant;
            if (-0.01f <= rat2 && rat2 <= 1.01f && -0.01f <= rat1 && rat1 <= 1.01f)
            {
                intersectionPoint = start + (rat1 * line1);
                return true;
            }
            intersectionPoint = Vector2.zero;
            return false;
        }
    }
